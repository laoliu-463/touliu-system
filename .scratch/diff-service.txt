+    static boolean isRefreshCredentialInvalid(Throwable exception) {
+        Throwable current = exception;
+        while (current != null) {
+            if (current instanceof QianchuanApiException apiException) {
+                if ("凭证无效".equals(apiException.getCategory())) {
+                    return true;
+                }
+                String apiCode = apiException.getApiCode();
+                if ("40102".equals(apiCode) || "40103".equals(apiCode) || "40105".equals(apiCode)
+                        || "HTTP_401".equals(apiCode)) {
+                    return true;
+                }
+            }
+            String message = current.getMessage();
+            if (message != null) {
+                String normalized = message.toLowerCase(Locale.ROOT);
+                if (normalized.contains("refresh_token") && (normalized.contains("invalid")
+                        || normalized.contains("expired") || normalized.contains("过期")
+                        || normalized.contains("无效") || normalized.contains("撤销"))) {

+                }
+                if (normalized.contains("access_token") && (normalized.contains("invalid")
+                        || normalized.contains("expired") || normalized.contains("过期")
+                        || normalized.contains("无效"))) {
+                    return true;
+                }
+                if (normalized.contains("授权已撤销") || normalized.contains("authorization revoked")
+                        || normalized.contains("token has been revoked")) {
+                    return true;
+                }
+            }
+            current = current.getCause();
+        }
+        return false;
+    }
